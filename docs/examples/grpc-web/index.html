gRPC-Web Browser Client

Protobuf-basierte API-Aufrufe direkt aus dem Browser

Was ist gRPC-Web?
gRPC-Web ermöglicht es, gRPC-Services direkt aus dem Browser aufzurufen. Die Protobuf-Definitionen werden zu JavaScript/TypeScript kompiliert, wodurch typsichere API-Aufrufe möglich sind.

Architektur: Browser → gRPC-Web Proxy (Envoy) → gRPC Server

// grpc-web-example.proto - Service Definition

service ExampleService {
    // Unary RPC - simple request/response
    rpc GetUser(GetUserRequest) returns (GetUserResponse);

    // Unary RPC - list with pagination
    rpc ListProjects(ListProjectsRequest) returns (ListProjectsResponse);

    // Unary RPC - create operation
    rpc CreateProject(CreateProjectRequest) returns (CreateProjectResponse);

    // Server streaming RPC (supported by gRPC-Web)
    rpc WatchProjects(WatchProjectsRequest) returns (stream ProjectUpdate);
}

message User {
    string id = 1;
    string name = 2;
    string email = 3;
    int64 created_at = 4;
}

message Project {
    string id = 1;
    string name = 2;
    string description = 3;
    string owner_id = 4;
    repeated string tag_ids = 5;
    ProjectStatus status = 6;
}
// Generated client usage (TypeScript)
import { ExampleServiceClient } from './generated/ExampleServiceClientPb';
import { GetUserRequest, ListProjectsRequest } from './generated/grpc-web-example_pb';

// Client erstellen
const client = new ExampleServiceClient('http://localhost:8080');

// Unary RPC aufrufen
async function getUser(userId: string) {
    const request = new GetUserRequest();
    request.setUserId(userId);

    return new Promise((resolve, reject) => {
        client.getUser(request, {}, (err, response) => {
            if (err) {
                reject(err);
                return;
            }
            // Response ist typsicher!
            const user = response.getUser();
            resolve({
                id: user.getId(),
                name: user.getName(),
                email: user.getEmail(),
                createdAt: user.getCreatedAt()
            });
        });
    });
}

// Server Streaming
function watchProjects(projectIds: string[], onUpdate: (update) => void) {
    const request = new WatchProjectsRequest();
    request.setProjectIdsList(projectIds);

    const stream = client.watchProjects(request, {});

    stream.on('data', (response) => {
        onUpdate({
            projectId: response.getProjectId(),
            updateType: response.getUpdateType(),
            project: response.getProject()?.toObject(),
            timestamp: response.getTimestamp()
        });
    });

    stream.on('error', (err) => console.error('Stream error:', err));
    stream.on('end', () => console.log('Stream ended'));

    return stream;
}
# Setup Anleitung für gRPC-Web

## 1. Protobuf Compiler installieren
npm install -g grpc-tools grpc_tools_node_protoc_ts

## 2. Proto zu JavaScript kompilieren
protoc -I=. \
  --js_out=import_style=commonjs:./generated \
  --grpc-web_out=import_style=typescript,mode=grpcwebtext:./generated \
  grpc-web-example.proto

## 3. Envoy Proxy konfigurieren (grpc-web proxy)
# envoy.yaml - siehe separate Konfigurationsdatei

## 4. Server starten (z.B. Node.js mit @grpc/grpc-js)
npm install @grpc/grpc-js @grpc/proto-loader

## 5. Im Browser verwenden
# Die generierten Dateien importieren:
# - grpc-web-example_pb.js (Message Klassen)
# - ExampleServiceClientPb.ts (Service Client)

## Alternative: grpcwebproxy (Go-basiert)
# https://github.com/nicholasjackson/grpc-web
grpcwebproxy --backend_addr=localhost:9090 \
             --run_tls_server=false \
             --allow_all_origins

## Wichtige Hinweise:
# - gRPC-Web unterstützt KEINE bidirektionalen Streams
# - Nur Unary und Server Streaming sind möglich
# - Ein Proxy (Envoy oder grpcwebproxy) ist erforderlich

Protobuf Datenextraktion

Die Protobuf-Dateien definieren die Datenstrukturen. Nach der Kompilierung stehen diese als typisierte JavaScript-Klassen zur Verfügung: 

// Aus der Response die Daten extrahieren:
const response = await client.listProjects(request);

// Protobuf → JavaScript Object
const projects = response.getProjectsList().map(proj => ({
    id: proj.getId(),
    name: proj.getName(),
    description: proj.getDescription(),
    ownerId: proj.getOwnerId(),
    tagIds: proj.getTagIdsList(),
    status: proj.getStatus()  // Enum-Wert
}));

// Oder direkt als Plain Object:
const projectsRaw = response.toObject().projectsList;
