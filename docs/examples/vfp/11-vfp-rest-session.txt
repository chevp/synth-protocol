================================================================================
VFP REST API SESSION
================================================================================

VFP Semantik über Standard HTTP REST und WebSocket.
Alle Features, volle Kompatibilität mit existierender Infrastruktur.

================================================================================
HTTP REST ENDPOINTS
================================================================================

FILE OPERATIONS
  GET     /vfp/{path}              Lese Resource
  PUT     /vfp/{path}              Schreibe Resource
  DELETE  /vfp/{path}              Lösche Resource
  GET     /vfp/{path}?list=true    Liste Verzeichnis
  HEAD    /vfp/{path}              Nur Metadaten
  POST    /vfp/{path}/_move        Verschiebe
  POST    /vfp/{path}/_copy        Kopiere

VERSION OPERATIONS
  POST    /vfp/{path}/_commit      Commit erstellen
  POST    /vfp/{path}/_diff        Versionen vergleichen
  GET     /vfp/{path}/_log         Versionshistorie
  POST    /vfp/{path}/_revert      Zurücksetzen
  POST    /vfp/_branch             Branch verwalten
  POST    /vfp/_checkout           Branch wechseln
  POST    /vfp/_merge              Branches mergen

QUERY OPERATIONS
  GET     /vfp/{path}/_select      Suchen/Filtern
  POST    /vfp/_batch              Mehrere Operationen

REALTIME (WebSocket)
  ws://host/vfp/realtime           Bidirektionaler Stream

  "gravity": 9.81,
  "spawn_point": [0, 10, 0]
}


-- WRITE (Create) --

Host: api.example.com
Content-Type: application/json
X-Vfp-Message: Add enemy configuration

{
  "max_enemies": 50,
  "spawn_rate": 2.5
}

ETag: "xyz789"

{
  "version": { "tick": 43, "hash": "xyz789" }
}


-- WRITE (Update with Optimistic Locking) --
PUT /vfp/scenes/level_01/config.json HTTP/1.1
Host: api.example.com
Content-Type: application/json
If-Match: "abc123"
X-Vfp-Message: Update gravity

{
  "name": "Level 01",
  "gravity": 10.0,
  "spawn_point": [0, 10, 0]
}

HTTP/1.1 200 OK
ETag: "newetag"
X-Vfp-Tick: 44


-- WRITE (Conflict) --
PUT /vfp/scenes/level_01/config.json HTTP/1.1
If-Match: "old_etag"

Content-Type: application/json

{
  "code": "CONFLICT",
  "message": "Resource was modified",
  "version": { "tick": 44, "hash": "newetag" }
}

-- SELECT (Filter) --
GET /vfp/entities/_select?filter=type%3Dplayer&sort=score:desc&limit=10 HTTP/1.1

HTTP/1.1 200 OK

{
  "results": [
    { "path": "/entities/player_001", "data": { "type": "player", "score": 1500 } },
    { "path": "/entities/player_042", "data": { "type": "player", "score": 1200 } }
  ],
  "total": 2,
  "has_more": false
}


-- BATCH --
POST /vfp/_batch HTTP/1.1
Content-Type: application/json

{
  "atomic": true,
  "operations": [
    { "id": "op1", "method": "PUT", "path": "/config/a.json", "body": "{\"key\": 1}" },
    { "id": "op2", "method": "PUT", "path": "/config/b.json", "body": "{\"key\": 2}" },
    { "id": "op3", "method": "DELETE", "path": "/config/old.json" }
  ]
}

HTTP/1.1 200 OK

{
  "results": [
    { "id": "op1", "status": 201 },
    { "id": "op2", "status": 201 },
    { "id": "op3", "status": 204 }
  ],
  "all_ok": true
}


================================================================================
REALTIME (WebSocket)
================================================================================

WebSocket URL: ws://api.example.com/vfp/realtime

Alle Nachrichten sind JSON mit folgendem Format:
{
  "id": "request-id",
  "op": "OPERATION",
  "payload": { ... }
}


-- CONNECT --
Client connects to ws://api.example.com/vfp/realtime

Server:
{
  "op": "CONNECTED",
  "payload": { "tick": 60, "branch": "main" }
}


-- WATCH --
Client:
{
  "id": "req-001",
  "op": "WATCH",
  "payload": {
    "patterns": ["/scenes/level_01/nodes/*"],
    "initial": true,
    "from_tick": 0
  }
}

Server:
{
  "id": "req-001",
  "op": "WATCH_OK",
  "payload": {
    "subscription_id": "sub-abc123",
    "tick": 60,
    "matched": 15
  }
}


-- DELTA (Server Push) --
Server:
{
  "op": "DELTA",
  "payload": {
    "tick": 61,
    "nodes": [
      {
        "path": "/scenes/level_01/nodes/player_001",
        "op": "update",
        "fields": [
          { "field": "position", "value": [10.5, 2.0, 15.3] }
        ]
      }
    ]
  }
}


-- PUSH (Client Input) --
Client:
{
  "id": "req-002",
  "op": "PUSH",
  "payload": {
    "path": "/scenes/level_01/nodes/player_001",
    "field": "position",
    "value": [11.0, 2.0, 15.5],
    "seq": 1001,
    "predicted": true
  }
}

Server (ACK):
{
  "id": "req-002",
  "op": "ACK",
  "payload": { "tick": 62, "seq": 1001 }
}


-- CORRECTION (Server rejects input) --
Server:
{
  "op": "CORRECTION",
  "payload": {
    "tick": 62,
    "path": "/scenes/level_01/nodes/player_001",
    "field": "position",
    "value": [10.8, 2.0, 15.4],
    "rejected_seq": 1001,
    "reason": "Collision detected"
  }
}


-- PULL (Full Snapshot) --
Client:
{
  "id": "req-003",
  "op": "PULL",
  "payload": {
    "path": "/scenes/level_01",
    "format": "json"
  }
}

Server:
{
  "id": "req-003",
  "op": "PULL_OK",
  "payload": {
    "tick": 62,
    "count": 250,
    "data": { ... full scene state ... }
  }
}


-- UNWATCH --
Client:
{
  "id": "req-004",
  "op": "UNWATCH",
  "payload": { "subscription_id": "sub-abc123" }
}

Server:
{
  "id": "req-004",
  "op": "UNWATCH_OK"
}


-- PING/PONG --
Client:
{
  "id": "req-005",
  "op": "PING",
  "payload": { "timestamp": 1705312200000 }
}

Server:
{
  "id": "req-005",
  "op": "PONG",

    "client_timestamp": 1705312200000,
    "server_timestamp": 1705312200005
  }
}


================================================================================
HTTP HEADERS REFERENZ
================================================================================

REQUEST HEADERS:
  X-Vfp-Branch       Current branch (default: main)
  X-Vfp-Tick         Read at specific tick
  X-Vfp-Message      Commit message for write operations
  If-Match           ETag for optimistic locking
  If-None-Match      Conditional GET (304 Not Modified)

RESPONSE HEADERS:
  X-Vfp-Tick         Current server tick
  X-Vfp-Branch       Current branch
  X-Vfp-Hash         Content hash
  ETag               For caching and locking
  X-Vfp-Total        Total count for list operations
  X-Vfp-Cursor       Pagination cursor
  X-Vfp-Modified     Last modification timestamp
